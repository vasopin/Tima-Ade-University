        function escapeHtml(text) {\r\n
            const map = {\r\n
                '&': '&amp;',\r\n
                '<': '&lt;',\r\n
                '>': '&gt;',\r\n
                '"': '&quot;',\r\n
                "'": '&#039;'\r\n
            };\r\n
            return String(text || '').replace(/[&<>"']/g, (m) => map[m]);\r\n
        }\r\n

        if (!todayBtn || !prevBtn || !nextBtn || !monthYearEl || !calendarGrid || !loadingEl || !errorEl || !dayEventsContainer || !eventsTitle || !eventsList || !noEventsEl) {\r\n
            return;\r\n
        }\r\n

        let currentDate = new Date();\r\n
        let allEvents = {};\r\n
        let selectedDay = null;\r\n
\r\n
        const eventColors = {\r\n

        async function initCalendar() {\r\n
            await fetchEvents();\r\n

        async function fetchEvents() {\r\n
            try {\r\n
                loadingEl.classList.remove('hidden');\r\n
                errorEl.classList.add('hidden');\r\n
\r\n
                const year = currentDate.getFullYear();\r\n
                const month = currentDate.getMonth() + 1;\r\n
                const response = await fetch(`http://localhost/api/calendar/events?year=${year}&month=${month}`);\r\n
                const data = await response.json();\r\n
\r\n
                allEvents = {};\r\n
                (data.events || []).forEach((event) => {\r\n
                    const startDate = new Date(event.start);\r\n
                    const dateStr = startDate.toISOString().split('T')[0];\r\n
\r\n
                    if (!allEvents[dateStr]) {\r\n
                        allEvents[dateStr] = [];\r\n
                    }\r\n
\r\n
                    allEvents[dateStr].push(event);\r\n
                });\r\n
\r\n
                loadingEl.classList.add('hidden');\r\n
            } catch (error) {\r\n
                console.error('Error fetching events:', error);\r\n
                loadingEl.classList.add('hidden');\r\n
                errorEl.classList.remove('hidden');\r\n
            }\r\n

        function renderCalendar() {\r\n
            const year = currentDate.getFullYear();\r\n
            const month = currentDate.getMonth();\r\n
\r\n
            monthYearEl.textContent = new Date(year, month, 1).toLocaleDateString('en-US', { month: 'long', year: 'numeric' });\r\n
            calendarGrid.innerHTML = '';\r\n
\r\n
            const firstDay = new Date(year, month, 1).getDay();\r\n
            const daysInMonth = new Date(year, month + 1, 0).getDate();\r\n
            const daysInPrevMonth = new Date(year, month, 0).getDate();\r\n
            const today = new Date();\r\n
            const isCurrentMonth = today.getFullYear() === year && today.getMonth() === month;\r\n
            const todayDate = isCurrentMonth ? today.getDate() : null;\r\n
\r\n
            for (let i = firstDay - 1; i >= 0; i--) {\r\n
                calendarGrid.appendChild(createDayElement(daysInPrevMonth - i, true));\r\n

\r\n
            for (let day = 1; day <= daysInMonth; day++) {\r\n
                const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;\r\n
                const isToday = isCurrentMonth && day === todayDate;\r\n
                const hasEvents = !!(allEvents[dateStr] && allEvents[dateStr].length > 0);\r\n
                calendarGrid.appendChild(createDayElement(day, false, isToday, hasEvents, dateStr));\r\n
            }\r\n
\r\n
            const totalCells = calendarGrid.children.length;\r\n
            const remainingCells = 42 - totalCells;\r\n
